Мета лабораторної роботи

Засвоєння теоретичного матеріалу та набуття практичних навичок реалізації й застосування абстрактних структур даних стек (LIFO — «останнім прийшов, першим вийшов») та черга (FIFO — «першим прийшов, першим вийшов»), а також порівняння двох способів їхньої реалізації — на основі масиву фіксованого розміру та на основі однозв'язного списку.

Постановка задачі
1. Кожен варіант завдання визначає прикладну задачу (наприклад, перевірку збалансованості дужок, обчислення постфіксного виразу тощо) та структуру(и) даних, потрібну для її розв'язання (стек, чергу, або обидві).

2. Для розв'язання задачі написати дві програми, що реалізують ту саму структуру даних двома способами:

2.1 на основі одновимірного масиву фіксованого розміру (з окремими змінними/індексами для вершини стека або голови й хвоста черги);
2.2 на основі однозв'язного списку (динамічне виділення пам'яті: вузол зі значенням і вказівником на наступний вузол).
Прикладна логіка (сама задача варіанта) в обох програмах має бути однаковою — відрізнятися має лише внутрішня реалізація стека/черги.

3. Обидві реалізації мають надавати єдиний узгоджений набір функцій, наприклад для стека: push, pop, peek, is_empty, is_full; для черги: enqueue, dequeue, peek_front, is_empty, is_full. Прикладна логіка (main і функції задачі) має викликати ці функції однаково незалежно від того, яка реалізація підключена — тобто дві програми відрізняються лише файлом/блоком, де реалізовані самі функції структури даних.

4. Розмір масиву для масивної реалізації взяти самостійно, достатнім для тестових прикладів (наприклад, 50–100 елементів), і оформити як іменовану константу.

5. Масивна реалізація черги обов'язково має бути кільцевою (циклічною): після заповнення останньої комірки масиву черга повинна продовжувати використовувати комірки, звільнені операціями dequeue, починаючи знову з початку масиву, а не вважати масив вичерпаним.

Варіант 6
Стеком обчислити значення постфіксного виразу з операціями + - * /, підтримкою унарного мінуса як окремої операції (наприклад, 5 neg 3 +) та коректною обробкою ділення на нуль.

Текст 1 програми:
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_SIZE 50
#define MAX_LINE 1024

typedef struct {
    double data[MAX_SIZE];
    int top;
} Stack;

void init(Stack *s) {
    s->top = -1;
}

int is_empty(Stack *s) {
    return s->top == -1;
}

int is_full(Stack *s) {
    return s->top == MAX_SIZE - 1;
}

int push(Stack *s, double value) {
    if (is_full(s)) {
        return 0;
    }

    s->top++;
    s->data[s->top] = value;

    return 1;
}

int pop(Stack *s, double *value) {
    if (is_empty(s)) {
        return 0;
    }

    *value = s->data[s->top];
    s->top--;

    return 1;
}

int peek(Stack *s, double *value) {
    if (is_empty(s)) {
        return 0;
    }

    *value = s->data[s->top];
    return 1;
}

int main() {
    Stack s;
    init(&s);

    char line[MAX_LINE];

    printf("Enter postfix expression: ");

    if (fgets(line, sizeof(line), stdin) == NULL) {
        printf("Error: no input.\n");
        return 1;
    }

    char *token = strtok(line, " \t\r\n");

    while (token != NULL) {

        if (strcmp(token, "neg") == 0) {
            double a;

            if (!pop(&s, &a)) {
                printf("Error: not enough operands.\n");
                return 1;
            }

            if (!push(&s, -a)) {
                printf("Error: stack is full.\n");
                return 1;
            }

        } else if (strcmp(token, "+") == 0 ||
                   strcmp(token, "-") == 0 ||
                   strcmp(token, "*") == 0 ||
                   strcmp(token, "/") == 0) {

            double a, b, result;

            if (!pop(&s, &b) || !pop(&s, &a)) {
                printf("Error: not enough operands.\n");
                return 1;
            }

            if (strcmp(token, "+") == 0) {
                result = a + b;
            } else if (strcmp(token, "-") == 0) {
                result = a - b;
            } else if (strcmp(token, "*") == 0) {
                result = a * b;
            } else {
                if (b == 0) {
                    printf("Error: division by zero.\n");
                    return 1;
                }

                result = a / b;
            }

            if (!push(&s, result)) {
                printf("Error: stack is full.\n");
                return 1;
            }

        } else {
            char *end;
            double number = strtod(token, &end);

            if (end == token || *end != '\0') {
                printf("Error: invalid token %s.\n", token);
                return 1;
            }

            if (!push(&s, number)) {
                printf("Error: stack is full.\n");
                return 1;
            }
        }

        token = strtok(NULL, " \t\r\n");
    }

    double answer;

    if (s.top != 0 || !peek(&s, &answer)) {
        printf("Error: incorrect expression.\n");
        return 1;
    }

    printf("Result: %.7f\n", answer);

    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    double data;
    struct Node *next;
} Node;

typedef struct {
    Node *top;
} Stack;

int is_empty(Stack *s) {
    return s->top == NULL;
}

int is_full(Stack *s) {
    (void)s;
    return 0;
}

int push(Stack *s, double x) {
    Node *p = malloc(sizeof(Node));

    if (p == NULL) return 0;

    p->data = x;
    p->next = s->top;
    s->top = p;

    return 1;
}

int pop(Stack *s, double *x) {
    if (is_empty(s)) return 0;

    Node *p = s->top;
    *x = p->data;
    s->top = p->next;
    free(p);

    return 1;
}

int peek(Stack *s, double *x) {
    if (is_empty(s)) return 0;

    *x = s->top->data;
    return 1;
}

void clear(Stack *s) {
    double x;
    while (pop(s, &x)) {}
}

int fail(Stack *s, const char *message) {
    printf("Error: %s\n", message);
    clear(s);
    return 1;
}

int main() {
    Stack s;
    s.top = NULL;

    char line[1024];
    printf("Enter postfix expression: ");

    if (!fgets(line, sizeof(line), stdin)) {
        return fail(&s, "no input");
    }

    char *token = strtok(line, " \t\r\n");

    while (token != NULL) {
        double a, b, value;

        if (strcmp(token, "neg") == 0) {
            if (!pop(&s, &a))
                return fail(&s, "not enough numbers");

            value = -a;
        }
        else if (strlen(token) == 1 &&
                 strchr("+-*/", token[0]) != NULL) {

            if (!pop(&s, &b) || !pop(&s, &a))
                return fail(&s, "not enough numbers");

            switch (token[0]) {
                case '+': value = a + b; break;
                case '-': value = a - b; break;
                case '*': value = a * b; break;
                case '/':
                    if (b == 0)
                        return fail(&s, "division by zero");

                    value = a / b;
                    break;
            }
        }
        else {
            char *end;
            value = strtod(token, &end);

            if (end == token || *end != '\0')
                return fail(&s, "invalid expression");
        }

        if (!push(&s, value))
            return fail(&s, "not enough memory");

        token = strtok(NULL, " \t\r\n");
    }

    double result;

    if (s.top == NULL || s.top->next != NULL ||
        !peek(&s, &result)) {
        return fail(&s, "incorrect expression");
    }

    printf("Result: %.7f\n", result);

    clear(&s);
    return 0;
}
```

отже, коли я виконувала лабораторну роботу, я використовувала стек та два способи його роботи. першу програму я створила за допомогою масиву, а другу за допомогою однозв'язного списку. обидві програми написані мовою С. протестувавши програми, я можу порівняти їх резцльтати. наприклад, в обох програмах я перевіряла математичні вирази з додаванням, множенням, діленням. результати в обох програмах були однакові. також я перевірила як програми реагують на помилки, наприклад якщо виконати операцію з порожнім стеком або поділити число на нуль, то програма виводить повідомлення про помилку. якщо порівнювати ці два способи, то основна різниця між ними це використання пам'яті. наприклад, в першій програмі максимальна кількість елементів дорівнює 50. коли я спробувала додати більше чисел, програма показала що стек запрвнений. а в другій програмі такого обмеження немає. з цього можна зробити висновок, що перший спосіб краще використовувати коли ми знаємо скільки елементів нам потрібно, а другий коли ця кількість може змінитися. також додавання та видалення елементів виконується за сталий час, тобто кількість елементів в стеку не впливає на кількість операцій які для цього потрібні. якщо сказати коротко, то обидві програми працюють правильно, але кожен спосіб має свої плюси і мінуси. тому не можна сказати що один спосіб завжди кращий за інший, все залежить від того для чого ми будемо використовувати стек.
